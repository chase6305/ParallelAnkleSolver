cmake_minimum_required(VERSION 3.16)
project(ParallelAnkleConsumer LANGUAGES CXX)

set(CONSUMER_FIND_MODE "core" CACHE STRING "core, config, all, optional-config, or an unsupported component")
if(CONSUMER_FIND_MODE STREQUAL "all")
    find_package(ParallelAnkle 0.1 CONFIG REQUIRED)
elseif(CONSUMER_FIND_MODE STREQUAL "optional-config")
    find_package(ParallelAnkle 0.1 CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS config)
else()
    find_package(ParallelAnkle 0.1 CONFIG REQUIRED COMPONENTS ${CONSUMER_FIND_MODE})
endif()

enable_testing()
if(CONSUMER_FIND_MODE STREQUAL "config")
    add_executable(consumer config.cpp)
    target_link_libraries(consumer PRIVATE ParallelAnkle::config)
    add_test(NAME installed_config COMMAND consumer "${ParallelAnkle_CONFIG_DIR}/rotary_ankle.yaml")
else()
    add_executable(consumer core.cpp)
    target_link_libraries(consumer PRIVATE ParallelAnkle::core)
    add_test(NAME installed_core COMMAND consumer)
endif()

if(CONSUMER_FIND_MODE STREQUAL "core" AND TARGET ParallelAnkle::config)
    message(FATAL_ERROR "An explicit core request unexpectedly imported the YAML loader")
endif()
if(CONSUMER_FIND_MODE STREQUAL "optional-config" AND CMAKE_DISABLE_FIND_PACKAGE_yaml-cpp)
    if(ParallelAnkle_config_FOUND OR TARGET ParallelAnkle::config)
        message(FATAL_ERROR "Optional config should remain unavailable when yaml-cpp discovery is disabled")
    endif()
endif()

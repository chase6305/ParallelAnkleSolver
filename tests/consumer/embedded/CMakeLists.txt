cmake_minimum_required(VERSION 3.16)
project(ParallelAnkleEmbeddedConsumer LANGUAGES CXX)

if(NOT DEFINED PARALLEL_ANKLE_SOURCE)
    message(FATAL_ERROR "Pass -DPARALLEL_ANKLE_SOURCE=<path to the source repository>")
endif()
set(PARALLEL_ANKLE_WITH_YAML OFF CACHE BOOL "Build an Eigen-only consumer" FORCE)
add_subdirectory("${PARALLEL_ANKLE_SOURCE}" parallel_ankle)
if(TARGET parallel_ankle_tests OR TARGET ankle_demo)
    message(FATAL_ERROR "add_subdirectory unexpectedly enabled tests or examples")
endif()
add_executable(consumer ../core.cpp)
target_link_libraries(consumer PRIVATE ParallelAnkle::core)
enable_testing()
add_test(NAME embedded_core COMMAND consumer)
